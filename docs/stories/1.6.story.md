**Backend Architecture Patterns**: [Source: architecture/backend-architecture.md]

```go
// Channel service for main lobby management
func (s *ChannelService) CreateMainLobby(roomID uint) (*Channel, error) {
    livekitRoomName := fmt.Sprintf("room_%d_main_lobby", roomID)

    mainLobby := &Channel{
        Name:            "Main Lobby",
        RoomID:          roomID,
        IsMainLobby:     true,
        LivekitRoomName: livekitRoomName,
    }

    if err := s.channelRepo.Create(mainLobby); err != nil {
        return nil, fmt.Errorf("failed to create main lobby: %w", err)
    }

    return mainLobby, nil
}

// User channel service for membership tracking
func (s *ChannelService) ConnectUserToMainLobby(userID uint, roomID uint) (*UserChannel, error) {
    mainLobby, err := s.channelRepo.FindMainLobbyByRoomID(roomID)
    if err != nil {
        return nil, fmt.Errorf("main lobby not found: %w", err)
    }

    // Generate LiveKit participant ID
    participantID := fmt.Sprintf("user_%d_%d", userID, time.Now().Unix())

    userChannel := &UserChannel{
        UserID:               userID,
        ChannelID:            mainLobby.ID,
        LivekitParticipantID: participantID,
    }

    if err := s.channelRepo.CreateUserChannel(userChannel); err != nil {
        return nil, fmt.Errorf("failed to connect user to main lobby: %w", err)
    }

    return userChannel, nil
}
```

**Database Schema Updates**: [Source: architecture/database-schema.md]
Required schema additions:
```sql
-- Channels table (implement via GORM AutoMigrate)
CREATE TABLE channels (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(50) NOT NULL,
    room_id INT UNSIGNED NOT NULL,
    is_main_lobby BOOLEAN DEFAULT FALSE,
    livekit_room_name VARCHAR(100) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (room_id) REFERENCES rooms(id) ON DELETE CASCADE,
    UNIQUE KEY unique_main_lobby_per_room (room_id, is_main_lobby),
    INDEX idx_room_channels (room_id),
    INDEX idx_livekit_room (livekit_room_name)
);

-- User channel tracking (implement via GORM AutoMigrate)
CREATE TABLE user_channels (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id INT UNSIGNED NOT NULL,
    channel_id INT UNSIGNED NOT NULL,
    connected_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    livekit_participant_id VARCHAR(100),

    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (channel_id) REFERENCES channels(id) ON DELETE CASCADE,
    UNIQUE KEY one_channel_per_user (user_id),
    INDEX idx_channel_users (channel_id)
);
```
